orders.js
// orders.js — today's order summary.
// It runs clean. Do not trust what it prints.

const orders = [
  { id: 1, item: "Keyboard",  qty: 2, price: 1200, status: "paid"      },
  { id: 2, item: "Mouse",     qty: 1, price: 650,  status: "cancelled" },
  { id: 3, item: "Monitor",   qty: 1, price: 9800, status: "paid"      },
  { id: 4, item: "USB Cable", qty: 3, price: 199,  status: "paid"      },
  { id: 5, item: "Headset",   qty: 1, price: 2400, status: "refunded"  },
];

function totalRevenue(orders) {
  let total = 0;
  for (let i = 1; i < orders.length; i++) {
    total += orders[i].price;
  }
  return total;
}

function averageOrderValue(orders) {
  return totalRevenue(orders) / orders.length;
}

function topSellingItem(orders) {
  let best = orders[0];
  for (const o of orders) {
    if (o.price > best.price) best = o;
  }
  return best.item;
}

console.log("Revenue:", totalRevenue(orders));
console.log("Average:", averageOrderValue(orders));
console.log("Top item:", topSellingItem(orders));
